cmake_minimum_required(VERSION 3.20)
project(XR1Play VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_package(SDL2 CONFIG QUIET)
find_package(SDL2_ttf CONFIG QUIET)
find_package(PkgConfig QUIET)
if(NOT TARGET SDL2::SDL2)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL2_LIBRARY NAMES SDL2 SDL2-2.0 REQUIRED)
  add_library(SDL2::SDL2 UNKNOWN IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${SDL2_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SDL2_INCLUDE_DIR}")
endif()
if(NOT TARGET SDL2_ttf::SDL2_ttf)
  find_path(SDL2_TTF_INCLUDE_DIR SDL_ttf.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL2_TTF_LIBRARY NAMES SDL2_ttf SDL2_ttf-2.0 REQUIRED)
  add_library(SDL2_ttf::SDL2_ttf UNKNOWN IMPORTED)
  set_target_properties(SDL2_ttf::SDL2_ttf PROPERTIES IMPORTED_LOCATION "${SDL2_TTF_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SDL2_TTF_INCLUDE_DIR}")
endif()
add_executable(xr1-play src/main.cpp)
target_compile_definitions(xr1-play PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(xr1-play PRIVATE SDL2::SDL2 SDL2_ttf::SDL2_ttf JPEG::JPEG Threads::Threads)
if(WIN32)
  set_target_properties(xr1-play PROPERTIES OUTPUT_NAME XR1-Play WIN32_EXECUTABLE TRUE)
  target_compile_definitions(xr1-play PRIVATE NOMINMAX _WIN32_WINNT=0x0602)
  target_link_libraries(xr1-play PRIVATE mfreadwrite mf mfplat mfuuid ole32 propsys uuid avrt)
  if(MSVC)
    target_link_options(xr1-play PRIVATE /ENTRY:mainCRTStartup)
  endif()
endif()
if(MSVC)
  target_compile_options(xr1-play PRIVATE /W4 /utf-8)
else()
  target_compile_options(xr1-play PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET xr1-play POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:xr1-play>/assets")
install(TARGETS xr1-play RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)

include(CTest)
if(BUILD_TESTING)
  add_executable(xr1-common-test tests/common_test.cpp)
  target_compile_definitions(xr1-common-test PRIVATE SDL_MAIN_HANDLED)
  target_link_libraries(xr1-common-test PRIVATE SDL2::SDL2 JPEG::JPEG Threads::Threads)
  if(MSVC)
    target_compile_options(xr1-common-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(xr1-common-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME common-codec-and-layouts COMMAND xr1-common-test)
  if(WIN32)
    foreach(kind IN ITEMS audio capture presenter theme)
      add_executable(xr1-windows-${kind}-test tests/windows_${kind}_test.cpp)
      target_compile_definitions(xr1-windows-${kind}-test PRIVATE SDL_MAIN_HANDLED NOMINMAX _WIN32_WINNT=0x0602)
      target_link_libraries(xr1-windows-${kind}-test PRIVATE SDL2::SDL2 JPEG::JPEG Threads::Threads mfreadwrite mf mfplat mfuuid ole32 propsys uuid avrt)
      if(MSVC)
        target_compile_options(xr1-windows-${kind}-test PRIVATE /UNDEBUG)
      else()
        target_compile_options(xr1-windows-${kind}-test PRIVATE -UNDEBUG)
      endif()
      if(kind STREQUAL "theme")
        target_link_libraries(xr1-windows-${kind}-test PRIVATE SDL2_ttf::SDL2_ttf)
      endif()
      add_test(NAME windows-${kind} COMMAND xr1-windows-${kind}-test)
    endforeach()
  endif()
endif()
