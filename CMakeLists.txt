cmake_minimum_required(VERSION 3.20)
project(XR1LPlayer VERSION 1.5.21 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_package(SDL2 CONFIG QUIET)
find_package(SDL2_ttf CONFIG QUIET)
find_package(PkgConfig QUIET)
if(NOT TARGET SDL2::SDL2)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL2_LIBRARY NAMES SDL2 SDL2-2.0 REQUIRED)
  add_library(SDL2::SDL2 UNKNOWN IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${SDL2_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SDL2_INCLUDE_DIR}")
endif()
if(NOT TARGET SDL2_ttf::SDL2_ttf)
  find_path(SDL2_TTF_INCLUDE_DIR SDL_ttf.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL2_TTF_LIBRARY NAMES SDL2_ttf SDL2_ttf-2.0 REQUIRED)
  add_library(SDL2_ttf::SDL2_ttf UNKNOWN IMPORTED)
  set_target_properties(SDL2_ttf::SDL2_ttf PROPERTIES IMPORTED_LOCATION "${SDL2_TTF_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SDL2_TTF_INCLUDE_DIR}")
endif()
add_executable(xr1-l-player src/main.cpp)
set_target_properties(xr1-l-player PROPERTIES OUTPUT_NAME xr1-l-player)
target_compile_definitions(xr1-l-player PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(xr1-l-player PRIVATE SDL2::SDL2 SDL2_ttf::SDL2_ttf JPEG::JPEG Threads::Threads)
if(WIN32)
  enable_language(RC)
  target_sources(xr1-l-player PRIVATE src/windows_app.rc)
  target_include_directories(xr1-l-player PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding" "${CMAKE_CURRENT_SOURCE_DIR}/src")
  set_source_files_properties(src/windows_app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/xr1-play.ico;${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/xr1-play-playstation.ico;${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/xr1-play-nintendo.ico;${CMAKE_CURRENT_SOURCE_DIR}/src/xr1-l-player.manifest")
  set_target_properties(xr1-l-player PROPERTIES OUTPUT_NAME XR1-L-Player WIN32_EXECUTABLE TRUE)
  target_compile_definitions(xr1-l-player PRIVATE NOMINMAX _WIN32_WINNT=0x0602)
  target_link_libraries(xr1-l-player PRIVATE mfreadwrite mf mfplat mfuuid ole32 propsys uuid avrt)
  if(MSVC)
    target_link_options(xr1-l-player PRIVATE /ENTRY:mainCRTStartup /MANIFEST:NO)
  endif()
endif()
if(MSVC)
  target_compile_options(xr1-l-player PRIVATE /W4 /utf-8)
else()
  target_compile_options(xr1-l-player PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET xr1-l-player POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:xr1-l-player>/assets")
install(TARGETS xr1-l-player RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)

  if(NOT WIN32)
    add_executable(xr1-menu-motion-test tests/menu_motion_test.cpp)

    foreach(kind IN ITEMS audio capture presenter theme ui motion icons)
